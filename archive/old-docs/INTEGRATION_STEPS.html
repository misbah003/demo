🎯 Step 8: Integration Examples

Example 1: Simple Form Integration

<form id="vatForm"> <select id="state"> <option value="Maharashtra">Maharashtra</option> <option value="Karnataka">Karnataka</option> <!-- ... all 36 states ... --> </select> <button type="submit">Calculate</button> </form> <script src="vat_predictor.js"></script> <script> document.getElementById('vatForm').addEventListener('submit', async (e) => { e.preventDefault(); const result = await VATPredictor.predict({ state: document.getElementById('state').value, // ... other fields ... }); alert('Refund: ' + VATPredictor.formatCurrency(result.predicted_refund_amount)); }); </script>

Example 2: Get All States Dynamically

<select id="state"></select> <script src="vat_predictor.js"></script> <script> // Populate dropdown with all states const states = VATPredictor.getStates(); const select = document.getElementById('state'); states.forEach(state => { const option = document.createElement('option'); option.value = state; option.textContent = state; select.appendChild(option); }); </script>
